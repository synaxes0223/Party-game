> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Avalon as a new game: full official role roster (Merlin/Percival/Morgana/Mordred/Oberon/Assassin), 5-10 players, Lady of the Lake for 7+ players, host-driven team building, reusing the reconnect capability built for Secret Mission Bingo.

**Architecture:** A pure-logic module `avalonLogic.js` (role dealing, compatibility validation, quest/vote/assassination resolution — no socket.io, no room state, mirrors `promptLogic.js`/`imposterLogic.js`) and a game module `avalon.js` (phase state machine wiring that logic into room state + socket emits, mirrors `secretMissions.js`'s reconnect hooks). `index.js` gains new socket handlers; host/player pages gain new screens following the existing per-game-prefixed-id convention (`av-*`).

**Tech Stack:** Node.js, Express, Socket.io v4, vanilla JS/HTML/CSS, Node's built-in test runner (`node:test`), `socket.io-client` for E2E, Playwright for the final browser playtest.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-12-avalon-design.md` — every requirement traces back to it.
- All server changes confined to `party-platform-full/party-game-platform/server/`.

Do not duplicate or modify that platform-level mechanism.

```js
// avalonLogic.js
// Pure functions for Avalon: role dealing + compatibility validation, quest
// team sizes, vote/quest resolution, Lady of the Lake eligibility, and
// assassination resolution. No socket.io, no room state -- plain data in,
// plain data out. See docs/superpowers/specs/2026-07-12-avalon-design.md.

const GOOD_EVIL_SPLIT = {
  5: { good: 3, evil: 2 },
  6: { good: 4, evil: 2 },
  7: { good: 4, evil: 3 },
  8: { good: 5, evil: 3 },
  9: { good: 6, evil: 3 },
  10: { good: 6, evil: 4 },
};

const QUEST_TEAM_SIZES = {
  5: [2, 3, 2, 3, 3],
  6: [2, 3, 4, 3, 4],
  7: [2, 3, 3, 4, 4],
  8: [3, 4, 4, 5, 5],
  9: [3, 4, 4, 5, 5],
  10: [3, 4, 4, 5, 5],
};

function shuffle(arr) {
  const copy = arr.slice();
  for (let i = copy.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [copy[i], copy[j]] = [copy[j], copy[i]];
  }
  return copy;
}

// Validates a host's role-config selection against the player count. Must
// be called (and pass) before dealRoles -- dealRoles trusts the evil/good
// special-role counts already fit within the headcount.
function validateConfig(playerCount, config) {
  const split = GOOD_EVIL_SPLIT[playerCount];
  if (!split) return { error: `Avalon supports 5-10 players (got ${playerCount}).` };
  if (config.ladyOfLake && playerCount < 7) {
    return { error: "Lady of the Lake requires at least 7 players." };
  }
  // Assassin is always present (the baseline evil special role, like Merlin
  // on the good side), so it always consumes one evil slot.
  const evilSpecialsCount = 1 + [config.mordred, config.morgana, config.oberon].filter(Boolean).length;
  if (evilSpecialsCount > split.evil) {
    return { error: `Too many evil special roles for ${playerCount} players (max ${split.evil} evil slots).` };
  }
  return {};
}

// Deals roles to playerIds according to config. Caller must have already
// called validateConfig(playerIds.length, config) and confirmed no error.
function dealRoles(playerIds, config) {
  const split = GOOD_EVIL_SPLIT[playerIds.length];

  const evilSpecials = ["assassin"];
  if (config.mordred) evilSpecials.push("mordred");
  if (config.morgana) evilSpecials.push("morgana");
  if (config.oberon) evilSpecials.push("oberon");

  const goodSpecials = ["merlin"];
  if (config.percival) goodSpecials.push("percival");

  const evilRoles = evilSpecials.concat(Array(split.evil - evilSpecials.length).fill("minion"));
  const goodRoles = goodSpecials.concat(Array(split.good - goodSpecials.length).fill("servant"));

  const allAssignments = shuffle(
    evilRoles.map((role) => ({ role, team: "evil" })).concat(goodRoles.map((role) => ({ role, team: "good" })))
  );

  const shuffledPlayers = shuffle(playerIds);
  const roles = new Map();
  shuffledPlayers.forEach((playerId, i) => roles.set(playerId, allAssignments[i]));
  return roles;
}

// Computes the private info a single player sees at role-reveal, per
// docs/superpowers/specs/2026-07-12-avalon-design.md section 1.
function computeRoleReveal(roles, playerId) {
  const me = roles.get(playerId);
  const reveal = { role: me.role, team: me.team };

  if (me.role === "merlin") {
    reveal.evilRoster = Array.from(roles.entries())
      .filter(([, r]) => r.team === "evil" && r.role !== "mordred")
      .map(([id]) => id);
  } else if (me.role === "percival") {
    reveal.merlinMorganaPair = shuffle(
      Array.from(roles.entries())
        .filter(([, r]) => r.role === "merlin" || r.role === "morgana")
        .map(([id]) => id)
    );
  } else if (me.team === "evil" && me.role !== "oberon") {
    reveal.evilTeammates = Array.from(roles.entries())
      .filter(([id, r]) => r.team === "evil" && r.role !== "oberon" && id !== playerId)
      .map(([id, r]) => ({ id, role: r.role }));
  }

  return reveal;
}

function getTeamSize(playerCount, questIndex) {
  return QUEST_TEAM_SIZES[playerCount][questIndex];
}

// The quest (0-based index) requiring 2 fails instead of 1 -- only quest 4
// (index 3) at 7+ players. Every other quest at every player count needs 1.
function failsRequiredFor(playerCount, questIndex) {
  if (playerCount >= 7 && questIndex === 3) return 2;
  return 1;
}

module.exports = {
  GOOD_EVIL_SPLIT,
  QUEST_TEAM_SIZES,
  shuffle,
  validateConfig,
  dealRoles,
  computeRoleReveal,
  getTeamSize,
  failsRequiredFor,
};
```

```js
function resolveVotes(activeIds, votes) {
  const approveCount = Array.from(votes.values()).filter(Boolean).length;
  const approved = approveCount > activeIds.length / 2; // strict majority; a tie rejects
  return { approved, approveCount, total: activeIds.length };
}

function resolveQuest(questCards, failsRequired) {
  const fails = Array.from(questCards.values()).filter((c) => c === "fail").length;
  return { fails, result: fails >= failsRequired ? "fail" : "success" };
}

function checkGameEnd(questHistory) {
  const successes = questHistory.filter((q) => q.result === "success").length;
  const fails = questHistory.filter((q) => q.result === "fail").length;
  if (successes >= 3) return { over: true, winner: "good", reason: "quests" };
  if (fails >= 3) return { over: true, winner: "evil", reason: "quests" };
  return { over: false };
}
```

```js
function lotlEligibleTargets(playerIds, holderId, everHeld) {
  return playerIds.filter((id) => id !== holderId && !everHeld.has(id));
}

function resolveLotlCheck(roles, targetId) {
  return roles.get(targetId).team; // "good" | "evil" -- alignment only, never the specific role
}

function resolveAssassination(roles, guessedPlayerId) {
  const target = roles.get(guessedPlayerId);
  const correct = Boolean(target && target.role === "merlin");
  return { correct, winner: correct ? "evil" : "good" };
```

```js
function onToggleTeamMember(room, io, playerId) {
  const gs = room.gameState;
  if (!gs || gs.phase !== "team-building") return { error: "Not building a team right now." };
  const idx = gs.proposedTeam.indexOf(playerId);
  if (idx === -1) {
    const teamSize = avalonLogic.getTeamSize(activePlayerIds(room).length, gs.questNumber - 1);
    if (gs.proposedTeam.length >= teamSize) return { error: "Team is already full." };
    gs.proposedTeam.push(playerId);
  } else {
    gs.proposedTeam.splice(idx, 1);
  }
  emitTeamBuildingState(room, io);
  return {};
}

function onProposeTeam(room, io) {
  const gs = room.gameState;
  if (!gs || gs.phase !== "team-building") return { error: "Not building a team right now." };
  const teamSize = avalonLogic.getTeamSize(activePlayerIds(room).length, gs.questNumber - 1);
  if (gs.proposedTeam.length !== teamSize) {
    return { error: `Team must have exactly ${teamSize} players (has ${gs.proposedTeam.length}).` };
  }
  gs.votes = new Map();
  gs.phase = "voting";
  io.in(room.code).emit("game:avalon-vote-open", {
    questNumber: gs.questNumber,
    team: gs.proposedTeam,
    leaderId: currentLeaderId(gs),
    rejectCount: gs.rejectCount,
  });
  return {};
}

function onVote(room, io, socketId, approve) {
  const gs = room.gameState;
  if (!gs || gs.phase !== "voting") return {};
  const activeIds = activePlayerIds(room);
  if (!activeIds.includes(socketId)) return {};
  gs.votes.set(socketId, Boolean(approve));
  io.to(room.hostSocketId).emit("game:avalon-vote-progress", { voted: gs.votes.size, total: activeIds.length });
  if (gs.votes.size >= activeIds.length) resolveVoteRound(room, io);
  return {};
}

function onForceResolveVotes(room, io) {
  const gs = room.gameState;
  if (!gs || gs.phase !== "voting") return { error: "Not currently voting." };
  for (const id of activePlayerIds(room)) {
    if (!gs.votes.has(id)) gs.votes.set(id, false);
  }
  resolveVoteRound(room, io);
  return {};
}

function resolveVoteRound(room, io) {
  const gs = room.gameState;
  const activeIds = activePlayerIds(room);
  const { approved } = avalonLogic.resolveVotes(activeIds, gs.votes);

  const voteList = activeIds.map((id) => ({
    id,
    nickname: room.players.get(id).nickname,
    approve: gs.votes.get(id),
  }));
  io.in(room.code).emit("game:avalon-vote-result", { votes: voteList, approved, team: gs.proposedTeam });

  if (approved) {
    gs.rejectCount = 0;
    gs.questCards = new Map();
    gs.phase = "questing";
    io.in(room.code).emit("game:avalon-quest-open", { questNumber: gs.questNumber, team: gs.proposedTeam });
    return;
  }

  gs.rejectCount += 1;
  if (gs.rejectCount >= 5) {
    gs.phase = "game-over";
    gs.winner = "evil";
    gs.winReason = "reject-streak";
    room.state = "results";
    io.in(room.code).emit("game:avalon-results", buildResultsPayload(room));
    return;
  }

  gs.leaderIndex += 1;
  gs.proposedTeam = [];
  gs.phase = "team-building";
  emitTeamBuildingState(room, io);
}
```

```js
    submitted: gs.questCards.size,
    total: gs.proposedTeam.length,
  });
  if (gs.questCards.size >= gs.proposedTeam.length) resolveQuestRound(room, io);
  return {};
}

function onForceResolveQuestCards(room, io) {
  const gs = room.gameState;
  if (!gs || gs.phase !== "questing") return { error: "Not currently questing." };
  for (const id of gs.proposedTeam) {
    if (!gs.questCards.has(id)) gs.questCards.set(id, "success");
  }
  resolveQuestRound(room, io);
  return {};
}

function resolveQuestRound(room, io) {
  const gs = room.gameState;
  const playerCount = activePlayerIds(room).length;
  const failsRequired = avalonLogic.failsRequiredFor(playerCount, gs.questNumber - 1);
  const { fails, result } = avalonLogic.resolveQuest(gs.questCards, failsRequired);

  gs.questHistory.push({ questNumber: gs.questNumber, team: gs.proposedTeam, fails, result });
  io.in(room.code).emit("game:avalon-quest-result", {
    questNumber: gs.questNumber,
    fails,
    result,
    questHistory: gs.questHistory,
  });

  const { over, winner, reason } = avalonLogic.checkGameEnd(gs.questHistory);
  if (over && winner === "evil") {
    gs.phase = "game-over";
    gs.winner = "evil";
    gs.winReason = reason;
    room.state = "results";
    io.in(room.code).emit("game:avalon-results", buildResultsPayload(room));
    return;
  }

  const justFinishedQuest = gs.questNumber;
  const needsLotl = gs.lotl && [2, 3, 4].includes(justFinishedQuest);

  if (over && winner === "good") {
    if (needsLotl) {
      gs.phase = "lady-of-lake";
      gs.lotl.nextPhase = "assassination";
      emitLotlTurn(room, io);
    } else {
      beginAssassination(room, io);
    }
    return;
  }

  gs.questNumber += 1;
  gs.proposedTeam = [];

  if (needsLotl) {
    gs.phase = "lady-of-lake";
    gs.lotl.nextPhase = "team-building";
    emitLotlTurn(room, io);
  } else {
    gs.phase = "team-building";
    emitTeamBuildingState(room, io);
  }
}

function buildResultsPayload(room, extra = {}) {
  const gs = room.gameState;
  const roles = Array.from(gs.roles.entries()).map(([id, r]) => ({
    id,
    nickname: room.players.get(id) ? room.players.get(id).nickname : "Unknown",
    role: r.role,
    team: r.team,
  }));
  return { winner: gs.winner, winReason: gs.winReason, roles, questHistory: gs.questHistory, ...extra };
}
```

```js
function onPlayerReconnected(room, io, oldSocketId, newSocketId) {
  const gs = room.gameState;
  if (!gs) return;

  if (gs.roles.has(oldSocketId)) {
    gs.roles.set(newSocketId, gs.roles.get(oldSocketId));
    gs.roles.delete(oldSocketId);
  }
  gs.leaderOrder = gs.leaderOrder.map((id) => (id === oldSocketId ? newSocketId : id));
  gs.proposedTeam = gs.proposedTeam.map((id) => (id === oldSocketId ? newSocketId : id));
  if (gs.votes.has(oldSocketId)) {
    gs.votes.set(newSocketId, gs.votes.get(oldSocketId));
    gs.votes.delete(oldSocketId);
  }
  if (gs.questCards.has(oldSocketId)) {
    gs.questCards.set(newSocketId, gs.questCards.get(oldSocketId));
    gs.questCards.delete(oldSocketId);
  }
  if (gs.assassinId === oldSocketId) gs.assassinId = newSocketId;
  if (gs.lotl) {
    if (gs.lotl.holderId === oldSocketId) gs.lotl.holderId = newSocketId;
    if (gs.lotl.everHeld.has(oldSocketId)) {
      gs.lotl.everHeld.delete(oldSocketId);
      gs.lotl.everHeld.add(newSocketId);
    }
  }

  onPlayerSync(room, io, newSocketId);
}

function onPlayerSync(room, io, socketId) {
  const gs = room.gameState;
  if (!gs) return;

  if (gs.roles.has(socketId)) {
    io.to(socketId).emit("game:avalon-role-reveal", avalonLogic.computeRoleReveal(gs.roles, socketId));

  if (gs.phase === "team-building") {
    emitTeamBuildingState(room, io);
  } else if (gs.phase === "voting") {
    io.to(socketId).emit("game:avalon-vote-open", {
      questNumber: gs.questNumber,
      team: gs.proposedTeam,
      leaderId: currentLeaderId(gs),
      rejectCount: gs.rejectCount,
    });
  } else if (gs.phase === "questing" && gs.proposedTeam.includes(socketId)) {
    io.to(socketId).emit("game:avalon-quest-open", { questNumber: gs.questNumber, team: gs.proposedTeam });
  } else if (gs.phase === "lady-of-lake" && socketId === gs.lotl.holderId) {

    io.to(socketId).emit("game:avalon-assassin-prompt", { roster });
  } else if (gs.phase === "game-over") {
    io.to(socketId).emit("game:avalon-results", buildResultsPayload(room));
```

```js
// test/e2e-avalon.js
// Live integration check: runs the real server in-process and drives full
// Avalon games through socket.io-client (no mocks). Run with:
// node test/e2e-avalon.js

const path = require("node:path");
const { io } = require("socket.io-client");

const PORT = 3103;
const URL = `http://localhost:${PORT}`;

function once(emitter, event) {
  return new Promise((resolve) => emitter.once(event, resolve));
}

function connect() {
  return new Promise((resolve) => {
    const s = io(URL);
    s.on("connect", () => resolve(s));
  });
}

async function createRoom() {
  const host = await connect();
  const created = await new Promise((resolve) => {
    host.once("host:room-created", resolve);
    host.emit("host:create-room");
  });
  return { host, roomCode: created.room.code };
}

async function joinPlayers(roomCode, names) {
  const players = [];
  for (const name of names) {
    const socket = await connect();
    await new Promise((resolve, reject) => {
      socket.once("player:joined", () => resolve());
      socket.once("player:join-error", (d) => reject(new Error(d.error)));
      socket.emit("player:join-room", { code: roomCode, nickname: name });
    });
    players.push({ name, socket });
  }
  return players;
}

async function selectGame(host, roomCode) {
  const selectedPromise = once(host, "room:game-selected");
  host.emit("host:select-game", { code: roomCode, gameId: "avalon" });
  await selectedPromise;
}

function assertTrue(condition, message) {
  if (!condition) throw new Error(`Assertion failed: ${message}`);
}

function findByRole(roleRevealByPlayer, role) {
  return Object.entries(roleRevealByPlayer).find(([, r]) => r.role === role)?.[0];
}

async function scenario1_fiveByPlayerFullGameToAssassination() {
  console.log("\n[Scenario 1] 5 players, one rejected proposal, straight to 3 quest successes and assassination");
  const { host, roomCode } = await createRoom();
  const players = await joinPlayers(roomCode, ["Alice", "Bob", "Carol", "Dave", "Eve"]);
  await selectGame(host, roomCode);

  const revealPromises = players.map((p) => once(p.socket, "game:avalon-role-reveal"));
  host.emit("host:avalon-configure-roles", { code: roomCode, config: { percival: true, morgana: true } });
  host.emit("host:avalon-start-game", { code: roomCode });
  const reveals = await Promise.all(revealPromises);
  const revealByPlayer = {};
  players.forEach((p, i) => (revealByPlayer[p.name] = reveals[i]));

  const merlinName = Object.keys(revealByPlayer).find((n) => revealByPlayer[n].role === "merlin");
  assertTrue(!!merlinName, "expected exactly one Merlin among the 5 players");
  const byNickname = (name) => players.find((p) => p.name === name);

  // Round 1: propose a team, everyone rejects, leader rotates
  const buildingPromise = once(host, "game:avalon-team-building");
  host.emit("host:avalon-toggle-team-member", { code: roomCode, playerId: players[0].socket.id });
  host.emit("host:avalon-toggle-team-member", { code: roomCode, playerId: players[1].socket.id });
  const voteOpenPromise = once(host, "game:avalon-vote-open");
  host.emit("host:avalon-propose-team", { code: roomCode });
  await voteOpenPromise;

  const rejectResultPromise = once(host, "game:avalon-vote-result");
  players.forEach((p) => p.socket.emit("player:avalon-vote", { code: roomCode, approve: false }));
  const rejectResult = await rejectResultPromise;
  assertTrue(rejectResult.approved === false, "expected the first proposal to be rejected");
  await buildingPromise.catch(() => {}); // may have already resolved from the initial build

  // Now approve every subsequent proposal and always play Success, through
  // 3 quests, tracking whichever players are proposed each time.
  for (let q = 0; q < 3; q++) {
    const building = await once(host, "game:avalon-team-building");
    const teamIds = players.slice(0, building.teamSize).map((p) => p.socket.id);
    const voteOpen = once(host, "game:avalon-vote-open");
    teamIds.forEach((id) => host.emit("host:avalon-toggle-team-member", { code: roomCode, playerId: id }));
    host.emit("host:avalon-propose-team", { code: roomCode });
    await voteOpen;

    const voteResultPromise = once(host, "game:avalon-vote-result");
    players.forEach((p) => p.socket.emit("player:avalon-vote", { code: roomCode, approve: true }));
    await voteResultPromise;

    const questResultPromise = once(host, "game:avalon-quest-result");
    players
      .filter((p) => teamIds.includes(p.socket.id))
      .forEach((p) => p.socket.emit("player:avalon-quest-card", { code: roomCode, card: "success" }));
    const questResult = await questResultPromise;
    assertTrue(questResult.result === "success", `expected quest ${q + 1} to succeed`);
  }

  const assassinBeginPromise = once(host, "game:avalon-assassination-begin");
  const begin = await assassinBeginPromise;
  const assassinPlayer = players.find((p) => p.socket.id === begin.assassinId);
  assertTrue(!!assassinPlayer, "expected an assassin to be identified");

  const resultsPromise = once(host, "game:avalon-results");
  const merlinId = byNickname(merlinName).socket.id;
  assassinPlayer.socket.emit("player:avalon-assassinate", { code: roomCode, targetId: merlinId });
  const results = await resultsPromise;
  assertTrue(results.winner === "evil", "expected a correct Merlin guess to flip the win to Evil");
  assertTrue(results.roles.length === 5, "expected a full 5-player role reveal");
  console.log("  PASS");

  host.close();
  players.forEach((p) => p.socket.close());
}
```

```js
// ---- Avalon ----
const AV_ROLE_LABELS = {
  merlin: "Merlin (Good)", percival: "Percival (Good)", servant: "Loyal Servant (Good)",
  mordred: "Mordred (Evil)", morgana: "Morgana (Evil)", oberon: "Oberon (Evil)",
  minion: "Minion of Mordred (Evil)", assassin: "Assassin (Evil)",
};

function nicknameFor(id) {
  const p = currentPlayers.find((pl) => pl.id === id);
  return p ? p.nickname : "Unknown";
}

let avMyTeam = null;

socket.on("game:avalon-role-reveal", (reveal) => {
  if (currentGameId !== "avalon") return;
  avMyTeam = reveal.team;
  document.getElementById("btn-av-fail").style.display = avMyTeam === "evil" ? "block" : "none";

  document.getElementById("av-role-name").textContent = AV_ROLE_LABELS[reveal.role] || reveal.role;
  const extra = document.getElementById("av-role-extra");
  extra.innerHTML = "";
  if (reveal.evilRoster) {
    extra.innerHTML = `<div class="sm-board-row">Evil players: ${reveal.evilRoster.map(nicknameFor).join(", ")}</div>`;
  } else if (reveal.merlinMorganaPair) {
    extra.innerHTML = `<div class="sm-board-row">Merlin & Morgana (in some order): ${reveal.merlinMorganaPair.map(nicknameFor).join(", ")}</div>`;
  } else if (reveal.evilTeammates) {
    extra.innerHTML = `<div class="sm-board-row">Fellow Evil: ${reveal.evilTeammates.map((t) => `${nicknameFor(t.id)} (${t.role})`).join(", ")}</div>`;
  }
  showScreen("avRoleReveal");
});

document.getElementById("btn-av-role-ack").addEventListener("click", () => {
  showScreen("avWaiting");
});

socket.on("game:avalon-vote-open", ({ team }) => {
  if (currentGameId !== "avalon") return;
  const list = document.getElementById("av-vote-team");
  list.innerHTML = team.map((id) => `<div class="sm-board-row">${nicknameFor(id)}</div>`).join("");
  document.getElementById("av-vote-status").textContent = "";
  document.querySelectorAll("#screen-av-vote button").forEach((b) => (b.disabled = false));
  showScreen("avVote");
});

function castAvVote(approve) {
  socket.emit("player:avalon-vote", { code: roomCode, approve });
  document.querySelectorAll("#screen-av-vote button").forEach((b) => (b.disabled = true));
  document.getElementById("av-vote-status").textContent = "Vote submitted — waiting for others…";
}
document.getElementById("btn-av-approve").addEventListener("click", () => castAvVote(true));
document.getElementById("btn-av-reject").addEventListener("click", () => castAvVote(false));

socket.on("game:avalon-quest-open", ({ team }) => {
  if (currentGameId !== "avalon" || !team.includes(myId)) return;
  document.getElementById("av-quest-status").textContent = "";
  document.querySelectorAll("#screen-av-quest button").forEach((b) => (b.disabled = false));
  showScreen("avQuest");
});

function submitAvQuestCard(card) {
  socket.emit("player:avalon-quest-card", { code: roomCode, card });
  document.querySelectorAll("#screen-av-quest button").forEach((b) => (b.disabled = true));
  document.getElementById("av-quest-status").textContent = "Card submitted — waiting for the rest of the team…";
}
document.getElementById("btn-av-success").addEventListener("click", () => submitAvQuestCard("success"));
document.getElementById("btn-av-fail").addEventListener("click", () => submitAvQuestCard("fail"));

socket.on("game:avalon-lotl-turn", ({ holderId, eligibleTargets }) => {
  if (currentGameId !== "avalon" || holderId !== myId) return;
  const container = document.getElementById("av-lotl-targets");
  container.innerHTML = "";
  eligibleTargets.forEach((id) => {
    const btn = document.createElement("button");
    btn.className = "vote-btn";
    btn.textContent = nicknameFor(id);
    btn.addEventListener("click", () => {
      socket.emit("player:avalon-lotl-check", { code: roomCode, targetId: id });
    });
    container.appendChild(btn);
  });
  showScreen("avLotl");
});

socket.on("game:avalon-lotl-result", ({ targetNickname, alignment }) => {
  if (currentGameId !== "avalon") return;
  document.getElementById("av-lotl-result-text").textContent =
    `${targetNickname} is ${alignment === "good" ? "Good 🛡️" : "Evil 🗡️"}`;
  showScreen("avLotlResult");
});

socket.on("game:avalon-assassination-begin", () => {
  if (currentGameId !== "avalon") return;
  if (document.getElementById("screen-av-assassin").classList.contains("active")) return;
  showScreen("avWaiting");
});

socket.on("game:avalon-assassin-prompt", ({ roster }) => {
  if (currentGameId !== "avalon") return;
  const container = document.getElementById("av-assassin-targets");
  container.innerHTML = "";
  roster
    .filter((p) => p.id !== myId)
    .forEach((p) => {
      const btn = document.createElement("button");
      btn.className = "vote-btn";
      btn.textContent = p.nickname;
      btn.addEventListener("click", () => {
        socket.emit("player:avalon-assassinate", { code: roomCode, targetId: p.id });
      });
      container.appendChild(btn);
    });
  showScreen("avAssassin");
});

socket.on("game:avalon-results", ({ winner, roles }) => {
  if (currentGameId !== "avalon") return;
  const youWon = roles.find((r) => r.id === myId)?.team === winner;
  document.getElementById("av-winner-text").textContent =
    (winner === "good" ? "🛡️ Good wins!" : "🗡️ Evil wins!") + (youWon ? " (you!)" : "");
  const list = document.getElementById("av-roles-reveal");
  list.innerHTML = "";
  roles.forEach((r) => {
    const row = document.createElement("div");
    row.className = `sm-board-row status-${r.team === "good" ? "claimed" : "busted"}`;
    const youTag = r.id === myId ? " (you)" : "";
    row.innerHTML = `<span>${r.nickname}${youTag}</span><span class="status">${r.role} (${r.team})</span>`;
    list.appendChild(row);
  });
  showScreen("avResults");
});
```
